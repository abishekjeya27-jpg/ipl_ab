<script>

    /* Correct answers */
    const correctAnswers = {
        q1: "img-tag",
        q2: "a-tag",
        q3: "table",
        q4: "h1",
        q5: "background"
    };


    /* Question text */
    const questionTexts = {
        q1: "1. Which tag is used to add an image?",
        q2: "2. Which tag is used to add a link?",
        q3: "3. Which tag is used to add a table?",
        q4: "4. Which tag is used to add font style?",
        q5: "5. Which tag is used to add colour?"
    };


    /* Elements */
    const quizForm = document.getElementById("quizForm");
    const errorBox = document.getElementById("errorBox");
    const quizSection = document.getElementById("quizSection");
    const resultsSection = document.getElementById("resultsSection");


    /* Submit quiz */
    quizForm.addEventListener("submit", function(event) {

        event.preventDefault();

        const username =
            document.getElementById("username").value.trim();

        let allAnswered = true;

        const userAnswers = {};


        /* Check all questions */
        for (let i = 1; i <= 5; i++) {

            const selectedOption =
                document.querySelector(
                    `input[name="q${i}"]:checked`
                );

            if (!selectedOption) {

                allAnswered = false;
                break;

            }

            userAnswers[`q${i}`] =
                selectedOption.value;
        }


        /* Validation */
        if (!username || !allAnswered) {

            errorBox.style.display = "block";

            window.scrollTo({
                top: 0,
                behavior: "smooth"
            });

            return;
        }


        errorBox.style.display = "none";


        let score = 0;

        const totalQuestions = 5;

        let reviewHTML = "";


        /* Calculate score */
        for (let i = 1; i <= 5; i++) {

            const questionKey = `q${i}`;

            const userAnswer =
                userAnswers[questionKey];

            const correctAnswer =
                correctAnswers[questionKey];

            const isCorrect =
                userAnswer === correctAnswer;


            if (isCorrect) {
                score++;
            }


            reviewHTML += `
                <div class="review-block ${
                    isCorrect
                    ? "correct-review"
                    : "incorrect-review"
                }">

                    <div class="review-q-text">
                        ${questionTexts[questionKey]}
                    </div>

                    <div class="review-details">

                        Your answer:
                        <strong>${userAnswer}</strong>

                        ${isCorrect ? "✅" : "❌"}

                        <br>

                        ${
                            !isCorrect
                            ? `Correct answer:
                               <strong>${correctAnswer}</strong>`
                            : "Correct answer!"
                        }

                    </div>

                </div>
            `;
        }


        /* Display score */
        document.getElementById("scoreContainer").innerHTML = `
            Hello, <strong>${username}</strong>!<br>
            Your Score: ${score} / ${totalQuestions}
            (${(score / totalQuestions) * 100}%)
        `;


        /* Reward */
        let badge;
        let title;
        let description;


        if (score === 5) {

            badge = "🏆";

            title = "HTML Grandmaster";

            description =
                "Perfect score! You answered every question correctly!";

            triggerConfetti();

        }

        else if (score >= 3) {

            badge = "🥇";

            title = "HTML Scholar";

            description =
                "Great effort! You have a strong grasp of HTML basics.";

        }

        else {

            badge = "📚";

            title = "HTML Novice";

            description =
                "Keep practicing! Review the correct answers below and try again.";

        }


        /* Reward box */
        document.getElementById("rewardBox").innerHTML = `

            <div class="reward-badge">
                ${badge}
            </div>

            <div class="reward-title">
                ${title}
            </div>

            <div class="reward-desc">
                ${description}
            </div>

        `;


        /* Review */
        document.getElementById("reviewContainer").innerHTML =
            reviewHTML;


        /* Switch pages */
        quizSection.style.display = "none";

        resultsSection.style.display = "block";


        /* Move screen to top */
        window.scrollTo({
            top: 0,
            behavior: "smooth"
        });

    });


    /* Reset quiz */
    function resetQuiz() {

        quizForm.reset();

        errorBox.style.display = "none";

        document
            .querySelectorAll(".confetti")
            .forEach(confetti => confetti.remove());

        resultsSection.style.display = "none";

        quizSection.style.display = "block";

        window.scrollTo({
            top: 0,
            behavior: "smooth"
        });

    }


    /* Confetti effect */
    function triggerConfetti() {

        const container =
            document.getElementById("quizContainer");


        const colors = [
            "#667eea",
            "#764ba2",
            "#f093fb",
            "#43a047",
            "#ffca28",
            "#ef5350",
            "#29b6f6"
        ];


        for (let i = 0; i < 45; i++) {

            const confetti =
                document.createElement("div");


            confetti.className = "confetti";


            confetti.style.left =
                Math.random() * 100 + "%";


            confetti.style.backgroundColor =
                colors[
                    Math.floor(
                        Math.random() * colors.length
                    )
                ];


            confetti.style.animationDuration =
                (Math.random() * 2 + 1.5) + "s";


            confetti.style.animationDelay =
                Math.random() * 0.5 + "s";


            confetti.style.transform =
                `rotate(${Math.random() * 360}deg)`;


            container.appendChild(confetti);

        }

    }

</script>
